                    coverMat.needsUpdate = true;
                },
                undefined,
                (err) => { console.warn("Failed to load local cover image."); }
            );

            function createBookGeo(w, l, d, r) {
                const shape = getBookShape(w, l, r);
                const geo = new THREE.ExtrudeGeometry(shape, { depth: d, bevelEnabled: true, bevelSegments: 3, steps: 1, bevelSize: 0.04, bevelThickness: 0.04 });
                geo.center(); geo.rotateX(-Math.PI / 2); return geo;
            }

            const cL = 14.0;
            
            // 1) The HTML .book has an exact aspect ratio of 1.25. 
            // So the total HTML width in 3D units is always cL * 1.25 (17.5).
            // The HTML page (half the book) is exactly 8.75 3D units wide.
            const pW = cL * 0.625; 
            
            // 2) Find pixels-to-units conversion to calculate the 20px padding
            const domBook = document.querySelector('.book');
            const pixelHeight = (domBook && domBook.clientHeight > 0) ? domBook.clientHeight : 800;
            const unitsPerPixel = cL / pixelHeight;
            
            // 3) The cover width (cW) should be the page width + 20px on the outer edge
            const cW = pW + (20 * unitsPerPixel);
            
            const cT = 0.15;

            const marginTop = cL * 0.04;
            const marginBottom = cL * 0.04 + (cL * 0.92 * 0.12);

            const safeScaleL = 0.98;
            const pL = (cL - marginTop - marginBottom) * safeScaleL;
            const pT = 0.25;

            const zOffset = (marginBottom - marginTop) / 2;

            const coverRadius = 0.4;
            const pageRadius = 0.3;
            const spineRadius = pT + cT / 2;

            const coverMatArray = [leatherMat, leatherMat, leatherMat, leatherMat, leatherMat, leatherMat];

            class SacredJournal {
                constructor() {
                    this.group = new THREE.Group();
                    this.isOpen = false;
                    this.currentAngle = 0;
                    this.targetAngle = 0;

                    this.currentScale = 1.0;
                    this.targetScale = 1.0;

                    this.currentPosY = 0;
                    this.targetPosY = 0;

                    this.currentRotX = 0.35;
                    this.targetRotX = 0.35;
                    this.currentRotY = -0.2;
                    this.targetRotY = -0.2;
                    this.targetPosX = -cW / 2;

                    const rightCover = new THREE.Mesh(createBookGeo(cW, cL, cT, coverRadius), coverMatArray);
                    rightCover.position.set(cW / 2, -pT - cT / 2, 0);
                    this.group.add(rightCover);

                    this.rightPageMat = new THREE.MeshBasicMaterial({ color: 0xffffff });
                    const rightPagesMatArray = [edgeMat, edgeMat, this.rightPageMat, edgeMat, edgeMat, edgeMat];
                    this.rightPages = new THREE.Mesh(createBookGeo(pW, pL, pT, pageRadius), rightPagesMatArray);
                    this.rightPages.position.set(pW / 2, -pT / 2, -zOffset);
                    this.group.add(this.rightPages);

                    this.spine = new THREE.Mesh(new THREE.CylinderGeometry(spineRadius, spineRadius, cL, 32, 1, false, Math.PI / 2, Math.PI), leatherMat);
                    this.spine.rotation.x = Math.PI / 2;
                    this.spine.position.set(0, -pT / 2, 0);
                    this.group.add(this.spine);

                    this.hinge = new THREE.Group();
                    this.hinge.position.set(0, 0, 0);
                    this.group.add(this.hinge);

                    const leftCover = new THREE.Mesh(createBookGeo(cW, cL, cT, coverRadius), coverMatArray);
                    leftCover.position.set(cW / 2, pT + cT / 2, 0);
                    this.hinge.add(leftCover);

                    const decalShape = getBookShape(cW - 0.2, cL - 0.2, coverRadius);
                    const decalGeo = new THREE.ShapeGeometry(decalShape);

                    decalGeo.computeBoundingBox();
                    const min = decalGeo.boundingBox.min;
                    const max = decalGeo.boundingBox.max;
                    const uvAttr = decalGeo.attributes.uv;
                    for (let i = 0; i < uvAttr.count; i++) {
                        uvAttr.setXY(i, (decalGeo.attributes.position.getX(i) - min.x) / (max.x - min.x), (decalGeo.attributes.position.getY(i) - min.y) / (max.y - min.y));
                    }

                    decalGeo.center();
                    decalGeo.rotateX(-Math.PI / 2);
                    const decal = new THREE.Mesh(decalGeo, coverMat);
                    decal.position.set(cW / 2, pT + cT + 0.05, 0);
                    this.hinge.add(decal);

                    this.leftPageMat = new THREE.MeshBasicMaterial({ color: 0xffffff });
                    const leftPagesMatArray = [edgeMat, edgeMat, edgeMat, this.leftPageMat, edgeMat, edgeMat];
                    this.leftPages = new THREE.Mesh(createBookGeo(pW, pL, pT, pageRadius), leftPagesMatArray);
                    this.leftPages.position.set(pW / 2, pT / 2, -zOffset);

                    this.leftPages.scale.set(-1, 1, 1);
                    this.hinge.add(this.leftPages);

                    this.group.rotation.x = this.currentRotX;
                    this.group.rotation.y = this.currentRotY;
                }

                update() {
                    const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
                    const speed = prefersReducedMotion ? 1.0 : (this.isOpen ? 0.06 : 0.15);

                    this.currentAngle += (this.targetAngle - this.currentAngle) * speed;
                    this.hinge.rotation.z = this.currentAngle;

                    this.currentRotX += (this.targetRotX - this.currentRotX) * speed;
                    this.group.rotation.x = this.currentRotX;
                    this.currentRotY += (this.targetRotY - this.currentRotY) * speed;
                    this.group.rotation.y = this.currentRotY;

                    const progress = this.currentAngle / Math.PI;
                    this.spine.scale.x = 1.0 - progress * (1.0 - (cT / spineRadius));
                    this.spine.position.y = -pT / 2 - progress * (pT / 2 + cT / 2);
                }
            }

            const journal3D = new SacredJournal();
            window.journal3D = journal3D;
            scene.add(journal3D.group);

            window.update3DPages = function () {
                if (window.game && game.pages && journal3D) {
                    try {
                        journal3D.leftPageMat.map = createDynamicPageTexture(game.pages[game.currentPageIdx], true);
                        journal3D.rightPageMat.map = createDynamicPageTexture(game.pages[game.currentPageIdx + 1], false);
                        journal3D.leftPageMat.needsUpdate = true;
                        journal3D.rightPageMat.needsUpdate = true;
                    } catch (e) { console.warn(e); }
                }
            };

            function alignCameraToDOM() {
                const domBook = document.querySelector('.book');
                if (!domBook || domBook.clientHeight === 0) return;

                const screenFraction = domBook.clientHeight / window.innerHeight;
                const visibleLengthAtZ = cL / screenFraction;
                const distance = visibleLengthAtZ / (2 * Math.tan(Math.PI * camera.fov / 360));

                const rect = domBook.getBoundingClientRect();
                const domCenterY = rect.top + rect.height / 2;
                const screenCenterY = window.innerHeight / 2;
                const pixelsPerUnit = domBook.clientHeight / cL;
                const cameraZOffset = -(domCenterY - screenCenterY) / pixelsPerUnit;

                camera.position.set(0, distance, -cameraZOffset);
                camera.lookAt(0, 0, -cameraZOffset);
            }

            const raycaster = new THREE.Raycaster();
            const mouse = new THREE.Vector2();

            container.addEventListener('pointerdown', (e) => {
                if (window.isDOMActive) return;
                mouse.x = (e.clientX / window.innerWidth) * 2 - 1;
                mouse.y = -(e.clientY / window.innerHeight) * 2 + 1;
                raycaster.setFromCamera(mouse, camera);
                if (raycaster.intersectObjects(journal3D.group.children, true).length > 0) {
                    window.initiateOpenSequence();
                }
            });

            window.isDOMActive = false;

            window.initiateOpenSequence = function () {
                if (journal3D.isOpen) return;
                journal3D.isOpen = true;
                journal3D.targetAngle = Math.PI;
                journal3D.targetRotX = 0;
                journal3D.targetRotY = 0;

                container.setAttribute('aria-expanded', 'true');
                const announcer = document.getElementById('sr-announcer');
                if (announcer) announcer.textContent = "Logbook opened.";
